create table public.timetables (
  id uuid not null default gen_random_uuid (),
  class_id uuid null,
  subject_name text not null,
  faculty_id text not null,
  day_of_week text not null,
  start_time time without time zone not null,
  end_time time without time zone not null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint timetables_pkey primary key (id),
  constraint timetables_class_id_fkey foreign KEY (class_id) references classes (id) on delete CASCADE,
  constraint timetables_faculty_id_fkey foreign KEY (faculty_id) references users (username) on delete set null
) TABLESPACE pg_default;
